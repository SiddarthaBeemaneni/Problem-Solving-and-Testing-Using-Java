class Result {

    /*
     * Complete the 'maxSubarray' function below.
     *
     * The function is expected to return an INTEGER_ARRAY.
     * The function accepts INTEGER_ARRAY arr as parameter.
     */

    public static List<Integer> maxSubarray(List<Integer> arr) {
    // Write your code here
     int currentSum = arr.get(0);
    int maxSubarray = arr.get(0);

    int maxSubsequence = arr.get(0);

    // Find maximum subarray
    for (int i = 1; i < arr.size(); i++) {

        currentSum = Math.max(arr.get(i),
                              currentSum + arr.get(i));

        maxSubarray = Math.max(maxSubarray, currentSum);
    }

    // Find maximum subsequence
    for (int i = 1; i < arr.size(); i++) {

        if (arr.get(i) > 0) {
            maxSubsequence += arr.get(i);
        }
    }

    // If all numbers are negative,
    // choose the largest (least negative) number
    if (maxSubsequence <= 0) {

        maxSubsequence = arr.get(0);

        for (int i = 1; i < arr.size(); i++) {
            maxSubsequence = Math.max(maxSubsequence, arr.get(i));
        }
    }

    return Arrays.asList(maxSubarray, maxSubsequence);

    }

}
