import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        ArrayList<ArrayList<Object>> a = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            String name = sc.next();
            double grade = sc.nextDouble();
            ArrayList<Object> x = new ArrayList<>();
            x.add(name); x.add(grade);
            a.add(x);
        }

        double min = Double.MAX_VALUE, second = Double.MAX_VALUE;
        for (ArrayList<Object> x : a) {
            double g = (double)x.get(1);
            if (g < min) { second = min; min = g; }
            else if (g > min && g < second) second = g;
        }

        ArrayList<String> names = new ArrayList<>();
        for (ArrayList<Object> x : a)
            if ((double)x.get(1) == second) names.add((String)x.get(0));

        Collections.sort(names);
        for (String name : names) System.out.println(name);
    }
}
