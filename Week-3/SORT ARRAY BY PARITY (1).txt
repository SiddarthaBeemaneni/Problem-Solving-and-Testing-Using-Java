class Solution {
    public int[] sortArrayByParity(int[] nums) {

        int[] result = new int[nums.length];
        int index = 0;

        // Store even numbers
        for (int num : nums) {
            if (num % 2 == 0) {
                result[index] = num;
                index++;
            }
        }

        // Store odd numbers
        for (int num : nums) {
            if (num % 2 != 0) {
                result[index] = num;
                index++;
            }
        }

        return result;
    }
}
